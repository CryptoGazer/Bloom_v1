# Loop Walk Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Tap 2–10 points on the phone map and get a closed walking or cycling route through them in tap order, back to the first point, shown like any route (two variants, reveal, details, navigator).

**Architecture:** The backend runs the existing A* leg by leg (1→2, …, N→1) for both variants, making segments already used twice as expensive so the way back takes other streets; the pieces are summarized once and judged by the same verdict as `/api/route` (extracted from `Network.plan`). A new `GET /api/loop` returns the `/api/route` shape plus `points` and `loop: true`. The phone page gets a loop mode on the start screen; the results, details and navigator screens are reused.

**Tech Stack:** Python 3.12, FastAPI, pytest; vanilla JS + Leaflet 1.9 in `web/m/`.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-11-15-loop-walk-design.md`.
- 2–10 points; order = tap order; the loop returns to point 1.
- Reuse penalty: segments already used by the loop cost ×2 for later legs (`SETTINGS["loop_reuse_factor"] = 2.0`).
- Walk and bike only (no transit in loop mode).
- No new explanatory labels or notes in the UI: the loop sheet holds a title, the walk/bike switch, two round icon buttons (undo, exit) and «Calculează».
- Romanian first; texts in `web/m/texts.js` for ro/en/ru.
- Tests run from `Bloom_v1/`: `../.venv/Scripts/python -m pytest backend/tests -q -p no:warnings` and `node --test web/m/nav.test.cjs`.

```python
    @classmethod
    def _verdict(cls, fastest, lit):
        """Whether the better-lit route is worth showing next to the fastest one (spec section 12).
        Both need "_key" and "_sec"."""
        gain = cls.lit_gain(fastest, lit)
        removes_dark_run = fastest["max_dark_run_m"] > SETTINGS["warn_dark_run_m"] >= lit["max_dark_run_m"]
        extra_min = (lit["_sec"] - fastest["_sec"]) / 60
        extra_share = (lit["_sec"] - fastest["_sec"]) / fastest["_sec"] if fastest["_sec"] else 0.0
        if fastest["_key"] == lit["_key"]:
            return {"kind": "same"}
        if gain < SETTINGS["same_if_gain_below"] and not removes_dark_run:
            return {"kind": "same", "near": True, "pick": "fastest", "lit_gain": round(gain, 3)}
        if extra_min <= SETTINGS["lit_free_extra_min"]:
            return {"kind": "same", "near": True, "pick": "better_lit", "lit_gain": round(gain, 3), "extra_min": round(extra_min, 1)}
        too_long = extra_min > SETTINGS["hide_if_extra_min"] or extra_share > SETTINGS["hide_if_extra_share"]
        return {"kind": "too_long" if too_long else "show", "extra_min": round(extra_min, 1), "extra_share": round(extra_share, 3)}

    def plan(self, A, B, mode="walk"):
        fastest, lit = self.route(A, B, mode=mode), self.route(A, B, SETTINGS["factor_better_lit"], mode=mode)
        if fastest is None or lit is None:
            return None
        verdict = self._verdict(fastest, lit)
        for r in (fastest, lit):
            del r["_key"], r["_sec"]
        return {"routes": {"fastest": fastest, "better_lit": lit}, "verdict": verdict}

    def loop(self, points, mode="walk"):
        """A closed route through snapped points in their order and back to the first, as plan() answers: the fastest
        and the better-lit loop and the verdict. Later legs pay loop_reuse_factor for segments the loop has already
        used, so the way back takes other streets where there are any. None if a leg has no route."""
        routes = {}
        for kind, factors in (("fastest", None), ("better_lit", SETTINGS["factor_better_lit"])):
            factor = [1.0] * len(STATUSES) if factors is None else [float(factors[s]) for s in STATUSES]
            pieces, used = [], {}
            for a, b in zip(points, points[1:] + points[:1]):
                if a.seg == b.seg and abs(a.t - b.t) < 1.0:
                    continue                                   # the same place twice in a row
                leg = self._search(a, b, factor, mode, avoid=used)
                if leg is None:
                    return None
                leg = [p for p in leg if p["len"] > 0.5] or leg[:1]
                for p in leg:
                    used[p["seg"]] = SETTINGS["loop_reuse_factor"]
                pieces.extend(leg)
            if not pieces:
                return None
            routes[kind] = self._summarize(pieces, mode)
        verdict = self._verdict(routes["fastest"], routes["better_lit"])
        for r in routes.values():
            del r["_key"], r["_sec"]
        return {"routes": routes, "verdict": verdict}
```

```python
    @app.get("/api/loop")
    def loop(points: str = Query(description="lat,lon;lat,lon;… in visiting order, 2 to 10 points; the loop returns to the first"),
             mode: str = Query("walk", description="'walk' or 'bike'"),
             at: str = Query("", description="leaving time, as for /api/route")):
        """A closed walk or ride through the points in their order and back to the first: the fastest and the
        better-lit loop, judged like /api/route. Later legs avoid streets the loop has already used."""
        if mode not in MODES:
            raise HTTPException(422, detail={"code": "mode_not_supported", "message": f"mode must be one of {', '.join(MODES)}"})
        raw = [p for p in points.split(";") if p.strip()]
        if not 2 <= len(raw) <= SETTINGS["loop_max_points"]:
            raise HTTPException(422, detail={"code": "bad_points", "message": f"2 to {SETTINGS['loop_max_points']} points as lat,lon;lat,lon"})
        net = network()
        snapped = []
        for i, value in enumerate(raw, 1):
            lat, lon = parse_point(value, f"point {i}")
            snap = net.snap(lat, lon, mode=mode)
            if snap is None:
                nearest = net.nearest_m(lat, lon, mode=mode)
                raise HTTPException(422, detail={"code": "outside_coverage", "point": i, "nearest_m": None if nearest is None else round(nearest),
                                                 "message": f"point {i} is farther than {SETTINGS['snap_max_m']:.0f} m from the mapped network"})
            snapped.append((lat, lon, snap))
        plan = net.loop([s for _, _, s in snapped], mode=mode)
        if plan is None:
            raise HTTPException(404, detail={"code": "no_route", "message": f"no {mode} loop through these points"})
        when = leaving_time(at)
        if state["signs"] is not None:
            for r in plan["routes"].values():
                r["signs"] = state["signs"].along(r, when)
        pts = [point(*s) for s in snapped]
        return {"mode": mode, "loop": True, "points": pts, "from": pts[0], "to": pts[0], **plan, "at": when.strftime("%Y-%m-%d %H:%M"),
                "settings": {**SETTINGS, "transit": TRANSIT, "lively": LIVELY, "live": LIVE}, "data_as_of": net.meta.get("as_of")}
```

```js
function computeLoop() {
  if (loopPts.length < 2) { go('home'); return; }
  const f = p => `${p.lat.toFixed(5)},${p.lng.toFixed(5)}`, req = ++routeReq;
  saveHash();
  stopAnim(); stopLive();
  plan = null; part = null;
  routeLayer.clearLayers(); markLayer.clearLayers(); lifeLayer.clearLayers(); vehLayer.clearLayers();
  go('results');
  fit(L.latLngBounds(loopPts).pad(0.2));
  getJSON(`/api/loop?points=${loopPts.map(f).join(';')}&mode=${mode}${at ? '&at=' + encodeURIComponent(at) : ''}`)
    .then(d => { if (req !== routeReq) return; plan = d; live = null; if (screen !== 'results') go('results', false); reveal(false); })
    .catch(err => {
      if (req !== routeReq) return;
      const d = err.detail || {};
      const text = d.code === 'outside_coverage' ? t('m.loopOutside', { n: d.point })
        : d.code === 'no_route' ? t('err.no_route_' + (mode === 'bike' ? 'bike' : 'walk')) : t('err.other', { code: d.code || err.message });
      go('home');
      toast(text);
    });
}
```
